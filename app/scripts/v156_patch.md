```ts
      // ─────────────────────────────────────────────────────────────
      // v156 HERO MATERIALS — sprite + shader + bloom trinity.
      // One painted master texture (strand_master.png), 5 tinted variants.
      // All use Constant lighting + Add blend + bloomThreshold=0 so they
      // glow over any ambient background (fixes the "invisible against
      // bright window" issue user reported in v155).
      // Surface shader: UV.y scrolls upward over time for the chiral
      // flowing-light effect; sin() flicker on alpha for organic pulse.
      // ─────────────────────────────────────────────────────────────
      const HERO_TINT: Record<string, string> = {
        danger:   '#ff3a28',  // crimson — urgent, hot
        supply:   '#3ce070',  // emerald-mint — resource, life
        junction: '#ff9020',  // amber-orange — crossroads, signal
        scenic:   '#5fa8ff',  // sky-blue — contemplative, calm
        cairn:    '#ffc850',  // gold — sacred monument
      };
      const heroSurfaceShader = {
        uniforms: 'uniform highp float u_time;\nuniform highp float u_speed;',
        body: `
          highp vec2 uv = _surface.diffuse_texcoord;
          highp float scroll = u_time * 0.0006 * u_speed;
          highp vec2 flowUv = vec2(uv.x, fract(uv.y - scroll));
          highp vec4 sampled = texture(diffuseTexture, flowUv);
          highp float flicker = 0.85 + 0.15 * sin(u_time * 0.005 + uv.y * 6.0);
          highp float tipFade = smoothstep(1.0, 0.6, uv.y);
          highp float baseFade = smoothstep(0.0, 0.15, uv.y);
          _surface.diffuse_color.rgb *= sampled.rgb * flicker;
          _surface.diffuse_color.a = sampled.a * flicker * tipFade * baseFade;
        `,
      } as any;
      for (const t of types) {
        const cap = t.charAt(0).toUpperCase() + t.slice(1);
        // Hero strand material — sprite + shader + tint
        mats[`hero${cap}`] = {
          lightingModel: 'Constant',
          diffuseTexture: STRAND_MASTER,
          diffuseColor: HERO_TINT[t],
          blendMode: 'Add',
          bloomThreshold: 0.0,
          writesToDepthBuffer: false,
          readsFromDepthBuffer: true,
          shaderModifiers: { surface: heroSurfaceShader },
          materialUniforms: [
            { name: 'u_time', type: 'float', value: 0 },
            { name: 'u_speed', type: 'float', value: t === 'cairn' ? 1.4 : t === 'scenic' ? 0.6 : 1.0 },
          ],
        } as any;
        // Ember sprite for tip particles — solid white, additive
        mats[`ember${cap}`] = {
          lightingModel: 'Constant',
          diffuseTexture: STRAND_EMBER,
          diffuseColor: HERO_TINT[t],
          blendMode: 'Add',
          bloomThreshold: 0.0,
        };
      }

      ViroMaterials.createMaterials(mats);
```

```ts
  // Drive shader time uniforms from JS (canonical Viro pattern).
  // v156: every hero<Type> material gets its u_time pushed every 32ms so
  // the surface shader can scroll UVs upward and flicker alpha. Junction
  // legacy material kept for fallback compatibility.
  useEffect(() => {
    const start = Date.now();
    const cap = type.charAt(0).toUpperCase() + type.slice(1);
    const heroMatName = `hero${cap}`;
    const id = setInterval(() => {
      const t = Date.now() - start;

        } catch {}
      }
    }, 32);
```

## 失败回退

若 hero shader 注册时 throw: `try/catch` 已包裹在 useEffect 里,失败会落到
`crashLogger.breadcrumb('ritualAR:material-error ...')`,UI 仍可用旧 material 兜底。
