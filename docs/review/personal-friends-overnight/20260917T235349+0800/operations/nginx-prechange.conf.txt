server {
    server_name api.yiiling.cn;
    client_max_body_size 20m;

    # Forward everything to the backend container.
    # certbot --nginx will inject SSL config into this server block when run.
    location / {
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 60s;
        proxy_connect_timeout 10s;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/api.yiiling.cn/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/api.yiiling.cn/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
server {
    if ($host = api.yiiling.cn) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name api.yiiling.cn;
    client_max_body_size 20m;
    return 404; # managed by Certbot


}
